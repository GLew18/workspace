export function createWordmark(): Wordmark {
  const root = document.createElement('span');
  root.className = 'ws-mark';
  root.innerHTML = `<span class="ws-mark-text">C</span>${STONE_SVG}<span class="ws-mark-text">balt</span>`;
  return { el: root };
}

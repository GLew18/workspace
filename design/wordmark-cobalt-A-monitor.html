Ready to paste: modified createWordmark() (app/src/ui/laurel.ts)

// Only the two text spans change. MONITOR_SVG, keyboardPath(), the Wordmark
// interface, and all geometry constants stay exactly as they are today.

// #region Public API: createWordmark() builds the header logo element
export interface Wordmark {
  el: HTMLElement;
}

export function createWordmark(): Wordmark {
  const root = document.createElement('span');
  root.className = 'ws-mark';
  root.innerHTML = `<span class="ws-mark-text">C</span>${MONITOR_SVG}<span class="ws-mark-text">balt</span>`;
  return { el: root };
}
// #endregion
